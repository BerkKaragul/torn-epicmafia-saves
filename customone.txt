/*
  Fork of chain-watch-widget-custom.user.js (v2.0.2) — adds custom-URL sound
  support so each alert can play a hosted MP3/WAV/OGG instead of (or alongside)
  the built-in synth chimes/siren.
 
  Custom-URL behavior:
    - Sound picker per alert has a new "Custom URL" option; select it and
      supply a URL in the adjacent input.
    - On script load AND on settings-save, all configured URLs are pre-fetched
      via GM_xmlhttpRequest (bypasses page CSP + CORS), decoded via
      AudioContext.decodeAudioData(), and the resulting AudioBuffers cached
      in memory. Alerts then play instantly with no per-fire network latency.
    - If a URL is unreachable or the file is not a supported audio format,
      the alert falls back to the built-in warning siren so you still get
      an audible signal.
    - `@connect *` grants Tampermonkey permission to fetch from any host.
      Narrow this to your specific hosting domain(s) if you prefer tighter
      permissions (e.g. `@connect torn-sounds.pages.dev`).
 
  Recommended hosting for the URLs:
    - Cloudflare Pages: push a repo with MP3s in `/public/`, get
      https://<project>.pages.dev/<file>.mp3 — free, HTTPS, global CDN.
    - Cloudflare R2 with public bucket + custom domain — good if you have
      many files and want a real storage backend.
    - Railway: overkill for static files unless you already have a service.
 
  Everything else is inherited from the v2.0.2 fork.
*/
 
(function () {
  "use strict";
 
  const SITE = "https://torn-epicmafia-saves.vercel.app";
  const POLL_MS = 12000;
  const SETTINGS_KEY = "cw_custom_settings_v1";
 
  // Default sound library — served from ButtChew's Cloudflare Worker. Users can
  // still override this URL in the settings modal ("Sound library" section)
  // to point at their own manifest; leaving the field blank falls back to this.
  const DEFAULT_LIBRARY_MANIFEST_URL = "https://chain-saver-sounds.buttchew.workers.dev/manifest.json";
 
  // Debug logging — toggled via Tampermonkey menu (avoids userscript-sandbox
  // gotchas where `CW_DEBUG = true` in DevTools console doesn't reach the
  // userscript context). Persisted so it survives page reloads.
  let CW_DEBUG_ENABLED = GM_getValue("cw_debug", false);
  const dlog = (...args) => { if (CW_DEBUG_ENABLED) console.log("[ChainWatch]", ...args); };
 
  // ============================================================
  // ===== SETTINGS =============================================
  // ============================================================
  const SOUND_OPTIONS = [
    { id: "none", label: "None (silent)" },
    { id: "chimeA", label: "Chime A (soft bell)" },
    { id: "chimeB", label: "Chime B (rising ping)" },
    { id: "siren", label: "Siren (warning intensity)" },
    { id: "sirenCritical", label: "Siren (critical intensity)" },
    { id: "library", label: "From library (pick a hosted MP3)" },
    { id: "url", label: "Custom URL (paste any URL)" },
  ];
 
  const DEFAULTS = {
    alerts: [
      {
        id: "warn1", label: "Warning 1 (1:30)", seconds: 90, enabled: true,
        onDuty:  { sound: "chimeA", soundUrl: "", libraryFile: "", volume: 0.5, loop: false },
        offDuty: { sound: "none",   soundUrl: "", libraryFile: "", volume: 0.5, loop: false },
      },
      {
        id: "warn2", label: "Warning 2 (1:00)", seconds: 60, enabled: true,
        onDuty:  { sound: "chimeB", soundUrl: "", libraryFile: "", volume: 0.5, loop: false },
        offDuty: { sound: "none",   soundUrl: "", libraryFile: "", volume: 0.5, loop: false },
      },
      {
        id: "critical", label: "Critical (0:30)", seconds: 30, enabled: true,
        onDuty:  { sound: "sirenCritical", soundUrl: "", libraryFile: "", volume: 0.7, loop: true  },
        offDuty: { sound: "chimeB",        soundUrl: "", libraryFile: "", volume: 0.5, loop: false },
      },
    ],
    playerNameOverride: "",
    vibration: true,
    libraryManifestUrl: DEFAULT_LIBRARY_MANIFEST_URL,
  };
 
  function loadSettings() {
    try {
      const raw = GM_getValue(SETTINGS_KEY, null);
      if (!raw) return JSON.parse(JSON.stringify(DEFAULTS));
      const s = typeof raw === "string" ? JSON.parse(raw) : raw;
      // Merge shallowly so new fields inherit defaults
      return {
        alerts: Array.isArray(s.alerts) && s.alerts.length ? s.alerts : DEFAULTS.alerts,
        playerNameOverride: s.playerNameOverride ?? "",
        vibration: s.vibration ?? true,
        // OR (not ??) so an empty string from an older-version save also falls back to the default
        libraryManifestUrl: s.libraryManifestUrl || DEFAULT_LIBRARY_MANIFEST_URL,
      };
    } catch (e) {
      console.warn("[ChainWatch Custom] settings load failed", e);
      return JSON.parse(JSON.stringify(DEFAULTS));
    }
  }
 
  function saveSettings(s) {
    try { GM_setValue(SETTINGS_KEY, s); } catch (e) { console.warn("[ChainWatch Custom] settings save failed", e); }
  }
 
  // ============================================================
  // ===== PLAYER NAME / ROTATION ===============================
  // ============================================================
  function detectMyName() {
    try {
      for (let i = 0; i < sessionStorage.length; i++) {
        const key = sessionStorage.key(i);
        if (key && key.startsWith("sidebarData") && !key.includes("undefined")) {
          const data = JSON.parse(sessionStorage.getItem(key));
          const n = data && data.user && data.user.name;
          if (n) return String(n);
        }
      }
    } catch {}
    return null;
  }
 
  function getMyName() {
    const override = (loadSettings().playerNameOverride || "").trim();
    if (override) return override;
    return detectMyName();
  }
 
  function isMyTurn(data) {
    if (!data || !data.turn) return false;
    const myName = getMyName();
    if (!myName) return false;
    return String(data.turn).trim().toLowerCase() === myName.trim().toLowerCase();
  }
 
  // ============================================================
  // ===== AUDIO ================================================
  // ============================================================
  let audioCtx = null;
  function audio() {
    try {
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      if (audioCtx.state === "suspended") audioCtx.resume();
      return audioCtx;
    } catch (e) { return null; }
  }
 
  // Silent tick — call from a user gesture to unlock the AudioContext
  function armAudio() {
    const c = audio();
    if (!c) return false;
    const osc = c.createOscillator();
    const g = c.createGain();
    g.gain.value = 0.0001;
    osc.connect(g).connect(c.destination);
    osc.start();
    osc.stop(c.currentTime + 0.01);
    return true;
  }
 
  // Chime A — soft warm single note + subtle 5th, ~500ms decay
  function playChimeA(vol) {
    const c = audio(); if (!c) return;
    vol = Math.max(0, Math.min(1, vol));
    const t0 = c.currentTime;
    const dur = 0.5;
 
    const build = (freq, level) => {
      const g = c.createGain();
      g.gain.setValueAtTime(0, t0);
      g.gain.linearRampToValueAtTime(vol * level, t0 + 0.02);
      g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
      const o = c.createOscillator();
      o.type = "sine";
      o.frequency.value = freq;
      o.connect(g).connect(c.destination);
      o.start(t0);
      o.stop(t0 + dur);
    };
    build(440, 1.0);   // A4
    build(660, 0.35);  // E5 (subtle 5th)
  }
 
  // Chime B — two rising triangle-wave pings, brighter and more clipped than A
  function playChimeB(vol) {
    const c = audio(); if (!c) return;
    vol = Math.max(0, Math.min(1, vol));
    const noteDur = 0.14;
    const gap = 0.06;
    const t0 = c.currentTime;
 
    const note = (freq, startAt) => {
      const g = c.createGain();
      g.gain.setValueAtTime(0, startAt);
      g.gain.linearRampToValueAtTime(vol, startAt + 0.015);
      g.gain.exponentialRampToValueAtTime(0.001, startAt + noteDur);
      const o = c.createOscillator();
      o.type = "triangle";
      o.frequency.value = freq;
      o.connect(g).connect(c.destination);
      o.start(startAt);
      o.stop(startAt + noteDur);
    };
    note(700, t0);
    note(1050, t0 + noteDur + gap);
  }
 
  // Air-raid siren — from the original, refactored to accept a volume argument
  function playSiren(critical, vol) {
    const c = audio(); if (!c) return;
    vol = vol == null ? (critical ? 0.6 : 0.42) : Math.max(0, Math.min(1, vol));
    const t0 = c.currentTime;
    const dur = critical ? 1.5 : 1.1;
    const sweeps = critical ? 3 : 2;
    const lowHz = critical ? 620 : 440;
    const highHz = critical ? 1750 : 1150;
 
    const master = c.createGain();
    master.gain.setValueAtTime(0, t0);
    master.gain.linearRampToValueAtTime(vol, t0 + 0.02);
    master.gain.setValueAtTime(vol, t0 + dur - 0.08);
    master.gain.linearRampToValueAtTime(0, t0 + dur);
 
    const tremolo = c.createGain();
    tremolo.gain.value = 1;
    const lfo = c.createOscillator();
    const lfoDepth = c.createGain();
    lfo.type = "square";
    lfo.frequency.value = critical ? 14 : 9;
    lfoDepth.gain.value = 0.5;
    lfo.connect(lfoDepth).connect(tremolo.gain);
 
    const shaper = c.createWaveShaper();
    const curve = new Float32Array(257);
    for (let i = 0; i < 257; i++) {
      const x = (i / 256) * 2 - 1;
      curve[i] = Math.tanh(x * 3);
    }
    shaper.curve = curve;
 
    tremolo.connect(shaper).connect(master).connect(c.destination);
 
    const voice = (type, detune, level) => {
      const osc = c.createOscillator();
      const g = c.createGain();
      osc.type = type;
      osc.detune.value = detune;
      g.gain.value = level;
      const step = dur / (sweeps * 2);
      osc.frequency.setValueAtTime(lowHz, t0);
      for (let i = 0; i < sweeps; i++) {
        const base = t0 + i * step * 2;
        osc.frequency.linearRampToValueAtTime(highHz, base + step);
        osc.frequency.linearRampToValueAtTime(lowHz, base + step * 2);
      }
      osc.connect(g).connect(tremolo);
      osc.start(t0);
      osc.stop(t0 + dur);
    };
    voice("sawtooth", 0, 0.5);
    voice("square", 30, 0.28);
    if (critical) voice("sawtooth", -1200, 0.22);
 
    lfo.start(t0);
    lfo.stop(t0 + dur);
  }
 
  function maybeVibrate(pattern) {
    if (!loadSettings().vibration) return;
    try { if (navigator.vibrate) navigator.vibrate(pattern); } catch {}
  }
 
  // ============================================================
  // ===== CUSTOM-URL AUDIO CACHE ===============================
  // ============================================================
  // Fetch → decode → cache → play. Fetch goes through GM_xmlhttpRequest so we
  // bypass Torn's page CSP + any CORS friction. Once decoded, the AudioBuffer
  // sits in memory keyed by URL; subsequent plays are instant.
  const _mp3Cache = new Map();     // url -> AudioBuffer
  const _mp3Inflight = new Map();  // url -> Promise<AudioBuffer>
  const _mp3Failed = new Map();    // url -> { at, error } (for surfacing in diagnostics)
 
  function fetchAndDecode(url) {
    if (!url) return Promise.reject(new Error("no url"));
    if (_mp3Cache.has(url)) return Promise.resolve(_mp3Cache.get(url));
    if (_mp3Inflight.has(url)) return _mp3Inflight.get(url);
 
    const promise = new Promise((resolve, reject) => {
      dlog("MP3 fetch start", url);
      GM_xmlhttpRequest({
        method: "GET",
        url,
        responseType: "arraybuffer",
        timeout: 15000,
        onload: (r) => {
          if (r.status < 200 || r.status >= 300) {
            reject(new Error("HTTP " + r.status));
            return;
          }
          const ctx = audio();
          if (!ctx) { reject(new Error("AudioContext unavailable")); return; }
          // decodeAudioData works on suspended contexts — no user-gesture needed
          ctx.decodeAudioData(
            r.response,
            (buf) => {
              _mp3Cache.set(url, buf);
              _mp3Failed.delete(url);
              dlog("MP3 decoded + cached", url, buf.duration.toFixed(2) + "s");
              resolve(buf);
            },
            (err) => reject(new Error("decode failed: " + (err && err.message ? err.message : err)))
          );
        },
        onerror: (err) => reject(new Error("network error: " + JSON.stringify(err))),
        ontimeout: () => reject(new Error("timeout")),
      });
    });
 
    _mp3Inflight.set(url, promise);
    promise.then(
      () => _mp3Inflight.delete(url),
      (err) => {
        _mp3Inflight.delete(url);
        _mp3Failed.set(url, { at: Date.now(), error: String(err) });
        console.warn("[ChainWatch] MP3 fetch/decode failed", url, err);
      }
    );
    return promise;
  }
 
  function playBuffer(buf, vol) {
    const c = audio(); if (!c || !buf) return;
    vol = Math.max(0, Math.min(1, vol));
    const src = c.createBufferSource();
    const g = c.createGain();
    g.gain.value = vol;
    src.buffer = buf;
    src.connect(g).connect(c.destination);
    src.start();
  }
 
  // Play a URL sound. If already cached: plays instantly. If not cached:
  // triggers a fetch and, once done, plays. To avoid mid-alert silence when
  // the cache misses, we also fire the fallback siren immediately.
  function playUrl(url, vol) {
    if (!url) return;
    if (_mp3Cache.has(url)) {
      playBuffer(_mp3Cache.get(url), vol);
      return;
    }
    // Fallback fires now so the user isn't silent while we fetch
    console.warn("[ChainWatch] URL not cached, playing fallback siren while fetching", url);
    playSiren(false, vol);
    fetchAndDecode(url).catch(() => { /* already logged */ });
  }
 
  // Called on script init AND after settings save — pulls every configured URL
  // into the cache in parallel so alerts fire without network latency.
  function prewarmMp3Cache() {
    const urls = new Set();
    const s = loadSettings();
    for (const a of s.alerts) {
      for (const mode of [a.onDuty, a.offDuty]) {
        const url = resolveModeUrl(mode);
        if (url) urls.add(url);
      }
    }
    for (const url of urls) {
      if (_mp3Cache.has(url) || _mp3Inflight.has(url)) continue;
      fetchAndDecode(url).catch(() => { /* logged in fetch */ });
    }
  }
 
  // ============================================================
  // ===== SOUND LIBRARY (MANIFEST) =============================
  // ============================================================
  // A JSON manifest at a user-configured URL lists available sounds by
  // display name + file URL (relative or absolute). Widget fetches it once
  // per session; settings modal populates a dropdown from it so the user
  // doesn't have to memorize filenames.
  //
  // Accepted manifest formats:
  //   { "sounds": [{ "name": "Ding 01", "url": "ding-01.mp3" }] }
  //   [{ "name": "...", "url": "..." }]
  //   ["ding-01.mp3", "warning.mp3"]   (name derived from filename)
  let _library = null;              // [{ name, url }, ...] resolved absolute
  let _libraryInflight = null;
  let _libraryError = null;
 
  function normalizeManifest(raw, manifestUrl) {
    let entries;
    if (Array.isArray(raw)) entries = raw;
    else if (raw && Array.isArray(raw.sounds)) entries = raw.sounds;
    else return [];
 
    return entries.map((e) => {
      let name, url;
      if (typeof e === "string") { name = e; url = e; }
      else if (e && typeof e === "object" && e.url) {
        name = e.name || String(e.url).split("/").pop();
        url = e.url;
      } else return null;
      try { url = new URL(url, manifestUrl).href; } catch { return null; }
      return { name, url };
    }).filter(Boolean);
  }
 
  function fetchLibrary(force) {
    const manifestUrl = loadSettings().libraryManifestUrl;
    if (!manifestUrl) {
      _library = null;
      _libraryError = null;
      return Promise.resolve(null);
    }
    if (_library && !force) return Promise.resolve(_library);
    if (_libraryInflight && !force) return _libraryInflight;
 
    _libraryInflight = new Promise((resolve) => {
      dlog("library fetch start", manifestUrl);
      GM_xmlhttpRequest({
        method: "GET",
        url: manifestUrl,
        timeout: 10000,
        onload: (r) => {
          try {
            if (r.status < 200 || r.status >= 300) throw new Error("HTTP " + r.status);
            const raw = JSON.parse(r.responseText);
            _library = normalizeManifest(raw, manifestUrl);
            _libraryError = null;
            dlog("library loaded", _library.length + " entries");
          } catch (err) {
            _libraryError = String(err);
            _library = null;
            console.warn("[ChainWatch] library manifest parse failed", err);
          }
          resolve(_library);
        },
        onerror: (err) => {
          _libraryError = "network error";
          _library = null;
          console.warn("[ChainWatch] library manifest fetch failed", err);
          resolve(null);
        },
        ontimeout: () => {
          _libraryError = "timeout";
          _library = null;
          resolve(null);
        },
      });
    });
 
    _libraryInflight.finally(() => { _libraryInflight = null; });
    return _libraryInflight;
  }
 
  // Resolve a mode object to an absolute URL for its configured sound source,
  // or null if the mode is silent / built-in / not yet resolvable.
  function resolveModeUrl(mode) {
    if (!mode) return null;
    if (mode.sound === "url") return (mode.soundUrl || "").trim() || null;
    if (mode.sound === "library") {
      const file = (mode.libraryFile || "").trim();
      if (!file) return null;
      const manifest = loadSettings().libraryManifestUrl;
      if (!manifest) return null;
      // libraryFile is stored as the ABSOLUTE URL (populated from the
      // manifest at pick time) so we don't need to re-resolve here.
      return file;
    }
    return null;
  }
 
  // Sound dispatch. mode is the alert's onDuty or offDuty object:
  //   { sound, soundUrl?, volume, loop? }
  // opts.bypassMaster=true allows local test-plays even if this tab isn't
  // the cross-tab audio master (used by settings modal previews).
  function playSound(mode, opts) {
    opts = opts || {};
    if (!opts.bypassMaster && !isAudioMaster()) {
      dlog("playSound blocked by isAudioMaster()=false", mode);
      return;
    }
    dlog("playSound dispatch", { sound: mode.sound, volume: mode.volume, bypassMaster: !!opts.bypassMaster });
    const vol = mode.volume;
    try {
      switch (mode.sound) {
        case "chimeA": playChimeA(vol); break;
        case "chimeB": playChimeB(vol); break;
        case "siren":  playSiren(false, vol); maybeVibrate([220, 120, 220]); break;
        case "sirenCritical": playSiren(true, vol); maybeVibrate([300, 90, 300, 90, 600]); break;
        case "url":     playUrl(mode.soundUrl, vol); break;
        case "library": playUrl(resolveModeUrl(mode), vol); break;
        case "none":
        default: break;
      }
    } catch (err) {
      console.error("[ChainWatch] sound dispatch threw", mode, err);
    }
  }
 
  // ============================================================
  // ===== CROSS-TAB AUDIO MASTER (from original) ===============
  // ============================================================
  let sirenOn = GM_getValue("cw_siren", false);
  const TAB_ID = Date.now() + "-" + Math.random();
  const peers = {};
  let bc = null;
  try { if (window.BroadcastChannel) bc = new BroadcastChannel("cw_siren_audio"); } catch (e) {}
 
  const myVis = () => (document.hidden ? 1 : 0);
  function olderThan(a, b) {
    const na = parseInt(a, 10);
    const nb = parseInt(b, 10);
    return na !== nb ? na < nb : a < b;
  }
  function isAudioMaster() {
    if (!sirenOn) return false;
    if (!bc) return true;
    const now = Date.now();
    const myV = myVis();
    for (const id in peers) {
      const p = peers[id];
      if (now - p.t > 3000) { delete peers[id]; continue; }
      if (p.vis < myV) return false;
      if (p.vis === myV && olderThan(id, TAB_ID)) return false;
    }
    return true;
  }
  function announce(type) {
    if (bc) bc.postMessage({ type: type, id: TAB_ID, vis: myVis() });
  }
  if (bc) {
    bc.onmessage = function (e) {
      const m = e.data || {};
      if (m.type === "beat" && m.id) peers[m.id] = { t: Date.now(), vis: m.vis || 0 };
      else if (m.type === "bye" && m.id) delete peers[m.id];
    };
    setInterval(() => announce("beat"), 1000);
    document.addEventListener("visibilitychange", () => announce("beat"));
    window.addEventListener("pagehide", () => announce("bye"));
  }
 
  // ============================================================
  // ===== ALERT ENGINE =========================================
  // ============================================================
  // Per-chain dedup — remembers which alerts have already fired this chain
  // so single-shot chimes don't re-fire every render tick.
  const _firedForChain = { chainId: null, fired: new Set() };
  let _activeLoop = null; // { alertId, timerId }
  let _tickCounter = 0;
 
  function stopActiveLoop() {
    if (_activeLoop && _activeLoop.timerId) clearInterval(_activeLoop.timerId);
    _activeLoop = null;
  }
 
  // On chain change (or first observation): pre-mark any already-crossed
  // thresholds as fired so opening a mid-chain tab doesn't retro-alarm.
  function resetFiredIfChainChanged(chainId, remaining, settings) {
    if (chainId === _firedForChain.chainId) return;
    _firedForChain.chainId = chainId;
    _firedForChain.fired.clear();
    if (remaining == null) return;
    for (const a of settings.alerts) {
      if (!a.enabled) continue;
      if (remaining <= a.seconds) {
        _firedForChain.fired.add(a.id + ":on");
        _firedForChain.fired.add(a.id + ":off");
      }
    }
  }
 
  function updateAlerts(data, live, remaining) {
    // Heartbeat — proves updateAlerts is being called each tick (behind debug flag,
    // so it's silent in normal operation). Sampled every 5 ticks to avoid spam.
    if (CW_DEBUG_ENABLED) {
      _tickCounter = (_tickCounter + 1) % 5;
      if (_tickCounter === 0) {
        dlog("tick", { sirenOn, live, remaining, chainId: data?.chain?.id ?? null });
      }
    }
    if (!sirenOn || !live) { stopActiveLoop(); return; }
 
    const chainId = (data && data.chain && data.chain.id) || null;
    const settings = loadSettings();
    resetFiredIfChainChanged(chainId, remaining, settings);
 
    const myTurn = isMyTurn(data);
 
    // Find the deepest loop-mode alert that's currently applicable
    let bestLoop = null;
 
    for (const a of settings.alerts) {
      if (!a.enabled) continue;
      if (remaining > a.seconds) continue;
      const mode = myTurn ? a.onDuty : a.offDuty;
      if (!mode || !mode.sound || mode.sound === "none") continue;
 
      if (mode.loop) {
        if (!bestLoop || a.seconds < bestLoop.alert.seconds) {
          bestLoop = { alert: a, mode };
        }
      } else {
        const firedKey = a.id + (myTurn ? ":on" : ":off");
        if (_firedForChain.fired.has(firedKey)) continue;
        _firedForChain.fired.add(firedKey);
        playSound(mode);
      }
    }
 
    if (bestLoop) {
      const cadence = bestLoop.mode.sound === "sirenCritical" ? 1700 : 2600;
      if (_activeLoop && _activeLoop.alertId === bestLoop.alert.id) {
        dlog("loop already running", { alertId: _activeLoop.alertId, remaining });
        return;
      }
      stopActiveLoop();
      const doFire = () => playSound(bestLoop.mode);
      dlog("loop starting", { alertId: bestLoop.alert.id, sound: bestLoop.mode.sound, cadence, remaining, myTurn });
      doFire();
      _activeLoop = {
        alertId: bestLoop.alert.id,
        timerId: setInterval(doFire, cadence),
      };

  // Drag — PointerEvent primary, mouse/touch fallback (unchanged from original)
  let drag = null;
  function noDrag(target) {
    return target.tagName === "A" || (target.closest && target.closest("[data-cw-nodrag]"));
  }
  function moveTo(px, py) {
    const maxL = Math.max(0, window.innerWidth - box.offsetWidth);
    const maxT = Math.max(0, window.innerHeight - box.offsetHeight);
    box.style.left = Math.min(Math.max(0, px - drag.x), maxL) + "px";
    box.style.top = Math.min(Math.max(0, py - drag.y), maxT) + "px";
  }
  function persist() {
    drag = null;
    box.style.cursor = "grab";
    GM_setValue("cw_pos", { top: box.offsetTop, left: box.offsetLeft });
  }
  if (window.PointerEvent) {
    box.addEventListener("pointerdown", function (e) {
      if (noDrag(e.target)) return;
      drag = { x: e.clientX - box.offsetLeft, y: e.clientY - box.offsetTop, id: e.pointerId };
      try { box.setPointerCapture(e.pointerId); } catch (_) {}
      box.style.cursor = "grabbing";
      e.preventDefault();
    });
    box.addEventListener("pointermove", function (e) {
      if (!drag || e.pointerId !== drag.id) return;
      moveTo(e.clientX, e.clientY);
      e.preventDefault();
    });
    const up = function () {
      if (!drag) return;
      try { box.releasePointerCapture(drag.id); } catch (_) {}
      persist();
    };
    box.addEventListener("pointerup", up);
    box.addEventListener("pointercancel", up);
  } else {
    const pt = (e) => (e.touches && e.touches[0] ? e.touches[0] : e);
    const start = function (e) {
      if (noDrag(e.target)) return;
      const p = pt(e);
      drag = { x: p.x - box.offsetLeft, y: p.y - box.offsetTop };
      box.style.cursor = "grabbing";
    };
    const move = function (e) {
      if (!drag) return;
      const p = pt(e);
      moveTo(p.x, p.y);
      if (e.cancelable) e.preventDefault();
    };
    box.addEventListener("mousedown", start);
    box.addEventListener("touchstart", start, { passive: true });
    window.addEventListener("mousemove", move);
    window.addEventListener("touchmove", move, { passive: false });
    window.addEventListener("mouseup", persist);
    window.addEventListener("touchend", persist);
    window.addEventListener("touchcancel", persist);
  }

  function fmt(sec) {
    sec = Math.max(0, Math.floor(sec));
    return Math.floor(sec / 60) + ":" + String(sec % 60).padStart(2, "0");
  }
 
  // Escape text for safe interpolation into innerHTML (defensive against
  // hostile/compromised API responses — the original had raw interpolation).
  function esc(s) {
    return String(s == null ? "" : s)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#39;");
  }

        setTimeout(() => { btn.textContent = origText; btn.disabled = false; }, 2500);
      }
    };
 
    // Per-mode "Test URL" buttons — reads sibling URL input, fetches + plays
    panel.querySelectorAll("[data-cw-test-url]").forEach(btn => {
      btn.addEventListener("click", () => {
        const target = btn.getAttribute("data-cw-test-url");
        const input = panel.querySelector(`input[data-cw-field="${target}"]`);
        testUrl(btn, input && input.value.trim());
      });
    });
 
    // Per-mode "Test library" buttons — reads sibling library select, fetches + plays
    panel.querySelectorAll("[data-cw-test-library]").forEach(btn => {
      btn.addEventListener("click", () => {
        const target = btn.getAttribute("data-cw-test-library");
        const sel = panel.querySelector(`select[data-cw-field="${target}"]`);
        testUrl(btn, sel && sel.value);
      });
    });
 
    // Toggle URL / library row visibility as user changes the sound select
    panel.querySelectorAll("select[data-cw-field]").forEach(sel => {
      const field = sel.getAttribute("data-cw-field");
      if (!/\.sound$/.test(field)) return;
      const urlRow     = panel.querySelector(`[data-cw-url-row="${field.replace(/\.sound$/, ".soundUrl")}"]`);
      const libraryRow = panel.querySelector(`[data-cw-library-row="${field.replace(/\.sound$/, ".libraryFile")}"]`);
      const sync = () => {
        if (urlRow)     urlRow.style.display     = sel.value === "url"     ? "grid" : "none";
        if (libraryRow) libraryRow.style.display = sel.value === "library" ? "grid" : "none";
      };
      sync();
      sel.addEventListener("change", sync);
    });
 
    // "Reload library" button — refetch manifest from the URL currently in the input
    panel.querySelector('[data-cw-action="refresh-library"]').addEventListener("click", async () => {
      const input = panel.querySelector('input[data-cw-field="libraryManifestUrl"]');
      const url = input && input.value.trim();
      const statusEl = panel.querySelector("#cw-library-status");
      if (!url) { statusEl.textContent = "(enter a URL first)"; return; }
      // Persist the URL immediately so fetchLibrary() reads the new value
      const next = JSON.parse(JSON.stringify(settings));
      next.libraryManifestUrl = url;
      saveSettings(next);
      _library = null; _libraryError = null; _libraryInflight = null;
      statusEl.textContent = "(loading…)";
      await fetchLibrary(true);
      closeModal();
      openSettingsModal(); // reopen with fresh library populated
    });
 
    // Kick off a library fetch on modal open if the URL is set but we haven't loaded yet
    if (settings.libraryManifestUrl && !_library && !_libraryInflight) {
      fetchLibrary().then(() => {
        // If the modal is still open, patch the library selects and status line in place
        if (!document.body.contains(modal)) return;
        const statusEl = panel.querySelector("#cw-library-status");
        if (statusEl) {
          statusEl.innerHTML = _library
            ? `Loaded ${_library.length} sound${_library.length === 1 ? "" : "s"} from library.`
            : (_libraryError ? `<span style="color:#f87171">Library load failed: ${esc(_libraryError)}</span>` : "(no library)");
        }
        panel.querySelectorAll("select[data-cw-field]").forEach(sel => {
          const field = sel.getAttribute("data-cw-field");
          if (!/\.libraryFile$/.test(field)) return;
          // Preserve current selection if it's still in the library
          const currentValue = sel.value;
          sel.innerHTML = librarySelectHtml(currentValue);
          if (currentValue && [..._library || []].some(s => s.url === currentValue)) sel.value = currentValue;
        });
      });
    }
 
    // Close / Cancel / Reset / Save handlers
    const closeModal = () => modal.remove();
    panel.querySelector('[data-cw-action="close"]').addEventListener("click", closeModal);
    panel.querySelector('[data-cw-action="cancel"]').addEventListener("click", closeModal);
    panel.querySelector('[data-cw-action="reset"]').addEventListener("click", () => {
      if (confirm("Reset all settings to defaults?")) {
        saveSettings(JSON.parse(JSON.stringify(DEFAULTS)));
        closeModal();
        openSettingsModal();
      }
    });
    panel.querySelector('[data-cw-action="save"]').addEventListener("click", () => {
      const next = JSON.parse(JSON.stringify(settings));
      panel.querySelectorAll("[data-cw-field]").forEach(el => {
        const path = el.getAttribute("data-cw-field").split(".");
        let value;
        if (el.type === "checkbox") value = el.checked;
        else if (el.type === "number" || el.type === "range") value = parseFloat(el.value);
        else value = el.value;
        // Walk the path, creating intermediate structure as needed
        let obj = next;
        for (let i = 0; i < path.length - 1; i++) {
          const k = /^\d+$/.test(path[i]) ? parseInt(path[i], 10) : path[i];
          obj = obj[k];
        }
        const lastK = path[path.length - 1];
        obj[/^\d+$/.test(lastK) ? parseInt(lastK, 10) : lastK] = value;
      });
      saveSettings(next);
      // Kick off pre-warm for any newly configured URLs so they're ready
      prewarmMp3Cache();
      closeModal();
    });
 
    // Click backdrop to close
    modal.addEventListener("click", (e) => {
      if (e.target === modal) closeModal();
    });
 
    document.body.appendChild(modal);
  }
})();
